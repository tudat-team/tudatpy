cmake_minimum_required(VERSION 3.16)
project(StationLoadingDataStandalone LANGUAGES CXX)
get_filename_component(REPOSITORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
add_library(tudat_station_loading_data STATIC "${REPOSITORY}/src/tudat/io/readStationLoadingData.cpp")
target_compile_features(tudat_station_loading_data PUBLIC cxx_std_17)
target_include_directories(tudat_station_loading_data PUBLIC "${REPOSITORY}/include")
find_path(BOOST_TEST_INCLUDE_DIR boost/test/included/unit_test.hpp REQUIRED)
add_executable(test_station_loading "${REPOSITORY}/tests/test_tudat/src/io/unitTestStationLoadingData.cpp")
target_include_directories(test_station_loading SYSTEM PRIVATE "${BOOST_TEST_INCLUDE_DIR}")
target_link_libraries(test_station_loading PRIVATE tudat_station_loading_data)
enable_testing()
add_test(NAME StationLoadingData COMMAND test_station_loading)
