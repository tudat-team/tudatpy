cmake_minimum_required(VERSION 3.20)
project(TudatInstalledGravityConsumer LANGUAGES CXX)
find_package(Tudat CONFIG REQUIRED)
if(NOT DEFINED Tudat_BUILD_WITH_OPENMP)
    message(FATAL_ERROR "Installed Tudat package does not report its OpenMP policy")
endif()
add_executable(gravity_consumer main.cpp)
target_compile_features(gravity_consumer PRIVATE cxx_std_17)
target_compile_definitions(gravity_consumer PRIVATE
    EXPECT_OPENMP=$<BOOL:${Tudat_BUILD_WITH_OPENMP}>)
target_link_libraries(gravity_consumer PRIVATE Tudat::tudat_gravitation Tudat::tudat_basic_mathematics)
enable_testing()
add_test(NAME installed_gravity COMMAND gravity_consumer)
